cmake_minimum_required(VERSION 3.28)
project(jk_obs LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(jk_process STATIC src/ProcessBinding.cpp)
target_include_directories(jk_process PUBLIC include)
target_compile_definitions(jk_process PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
if(MSVC)
  target_compile_options(jk_process PRIVATE /W4 /WX)
endif()
include(CTest)
add_executable(ProcessBindingTests tests/ProcessBindingTests.cpp)
target_link_libraries(ProcessBindingTests PRIVATE jk_process)
foreach(name RejectsPidReuse SealedBindingNeverReopens ExitDuringActivationCannotEmit AccessDeniedDoesNotFallback RealProcessExitCannotRebind)
  add_test(NAME ProcessBinding.${name} COMMAND ProcessBindingTests ${name})
endforeach()
