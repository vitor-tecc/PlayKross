cmake_minimum_required(VERSION 3.28)
project(jk_instrumented_obs LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(libobs-winrt MODULE "${JK_PATCHED_SOURCE}/winrt-capture.cpp" "${JK_PATCHED_SOURCE}/winrt-dispatch.cpp")
target_include_directories(libobs-winrt PRIVATE "${JK_PATCHED_SOURCE}" "${JK_OBS_SOURCE}/libobs" "${JK_GENERATED_HEADERS}")
target_compile_definitions(libobs-winrt PRIVATE UNICODE _UNICODE NOMINMAX)
target_precompile_headers(libobs-winrt PRIVATE <d3d11.h> <DispatcherQueue.h> <dwmapi.h> <obs-module.h> <util/windows/ComPtr.hpp> <Windows.Graphics.Capture.Interop.h> <windows.graphics.directx.direct3d11.interop.h> <winrt/Windows.Foundation.Metadata.h> <winrt/Windows.Graphics.Capture.h> <winrt/Windows.System.h>)
target_link_libraries(libobs-winrt PRIVATE "${JK_OBS_IMPORT}/obs.lib" Dwmapi windowsapp)
set_target_properties(libobs-winrt PROPERTIES PREFIX "" OUTPUT_NAME "libobs-winrt")
